\section{Mathematical statements}
\begin{exercise}
    What is the negation of the statement ``either $X$ is true or $Y$ is true, but not both''?
\end{exercise}
\begin{solution}
$X$ and $Y$ are false or $X$ and $Y$ are true. 
\end{solution}

\begin{exercise}
    What is the negation of the statement ``$X$ is true if and only if $Y$ is true''?
\end{exercise}
\begin{solution}
    $X$ is false iff $Y$ is false
\end{solution}

\begin{exercise}
    Suppose that you have shown that whenever $X$ is true, then $Y$ is true, and whenever $X$ is false, then $Y$ is false. Have you demonstrated that $X$ and $Y$ are logically equivalent? Explain.
\end{exercise}
\begin{solution}
    A well-defined statement can be either true or false only (all possible values). Now, we consider the assumptions: (1) if $X$ is false, then $Y$ is false ($Y$ is never true in that case), and (2) if $X$ is true, then $Y$ is true ($Y$ is never false here). Thus, both statements coincide in all possible values what makes them ``equivalent''.
\end{solution}

\begin{exercise}
    Suppose that you have shown that whenever $X$ is true, then $Y$ is true, and whenever $Y$ is false, then $X$ is false. Have you demonstrated that $X \Leftrightarrow Y$? Explain.
\end{exercise}
\begin{solution}
    No. If $X$ is true, then $Y$ is true: we exclude only the case $X$ is true and $Y$ is false. If $Y$ is false, then $X$ is false: we exclude the case $X$ is true and $Y$ is false. The both cases represent the same statement. As the result, the case $X$ is False and $Y$ is True is missing. Thus, $X$ and $Y$ are not being proven to be logically equivalent.
\end{solution}

\begin{exercise}
    Suppose that you have shown that $X$ is true iff $Y$ is true, and you know that $Y$ is true iff $Z$ is true. Is this enough to show that $X, Y, Z$ are logically equivalent? Explain.
\end{exercise}
\begin{solution}
    Yes. Let's construct a logical table of all possible cases:
    \begin{center}
        \begin{tabular}{ccccc}
            $X$ & $Y$ & $Z$ & Possible? & If not, why?\\
            \hline
            False & False & False & Yes\\
            False & False & True & No & $Y \Leftrightarrow Z$\\
            False & True & False & No & $X \Leftrightarrow Y$ or $Y \Leftrightarrow X$\\
            False & True & True & No & $X \Leftrightarrow Y$\\
            True & False & False & No & $X \Leftrightarrow Y$\\
            True & False & True & No & $X \Leftrightarrow Y$ or $Y \Leftrightarrow X$\\
            True & True & False & No & $Y \Leftrightarrow Z$\\
            True & True & True & Yes\\
        \end{tabular}
    \end{center}
    What results $X, Y, Z$ to be logically equivalent by definition.
\end{solution}
\begin{exercise}
    Suppose that you have shown that whenever $X$ is true, then $Y$ is true; that if $Y$ is true, then $Z$ is true; and if $Z$ is true, then $X$ is true. Is this enough to show that $X, Y, Z$ are logically equivalent? Explain.
\end{exercise}
\begin{solution}
    Yes. Let's construct a logical table of all possible cases:
    \begin{center}
        \begin{tabular}{ccccc}
            $X$ & $Y$ & $Z$ & Possible? & If not, why?\\
            \hline
            False & False & False & Yes\\
            False & False & True & No & $Y \implies Z$\\
            False & True & False & No & $X \implies Y$\\
            False & True & True & No & $Z \implies X$\\
            True & False & False & No & $X \implies Y$\\
            True & False & True & No & $X \implies Y$\\
            True & True & False & No & $Y \implies Z$\\
            True & True & True & Yes\\
        \end{tabular}
    \end{center}
    What results $X, Y, Z$ to be logically equivalent by definition.
\end{solution}

\section{Implication}

The main ideas in the chapter are the following:
\begin{itemize}
    \item Vacuous hypothesis is a hypothesis where an antecedent is false. Since false implies false/true is always true, it does not provide any new information. 
    \item We can also draw the following conclusions:
    \begin{align*}
        X \implies Y & \iff \neg Y \implies \neg X & \text{Contrapositive}\\
        X \implies Y & \notiff Y \implies X & \text{Converse}\\
        X \implies Y & \notiff \neg X \implies \neg Y & \text{Inverse}
    \end{align*}
    \item Although, vacuous hypothesis can be viewed as ``useless'', it can still be used for a proof by contradiction using Contrapositive property, for instance. For that case, we need to have a consequent of the original implication to be known to be false. 
\end{itemize}

\begin{subexercises}
    \begin{exercise}
        For every positive number $x$, and every positive number $y$, we have $y^2 =x$.
    \end{exercise}
    \begin{solution}
        For any real number $x$ and $y$, $y^2=x$. False. Let's prove by counter example. If $x=1, y=2$, then $2^2\contreq1$. A game metaphor would be ``Your opponent picks real numbers $x, y$. If you can show that $y^2=x$ in every game, you win''.
    \end{solution}

    \begin{exercise}
        There exists a positive number $x$ such that for every positive number $y$, we have $y^2 = x$.
    \end{exercise}
    \begin{solution}
        One can pick a positive number $x$ s.t. $y^2=x$ for any positive $y$. False. Let's assume that such $x$ exists and we pick two positive numbers $y_1$ and $y_2$, for instance, 2 and 3 respectively. Then we have $y_1^2=4=x$ and $y_2^2=9=x$. If $4=x$ and $9=x$, then $4\contreq9$. A game metaphor would be ``You pick a positive number $x$. Then an opponent pick a positive number $y$. If you can show that $y^2=x$ in every game, you win''.
    \end{solution}


    \begin{exercise}
        There exists a positive number $x$, and there exists a positive number $y$, such that $y^2 = x$.
    \end{exercise}
    \begin{solution}
        One can pick two positive numbers $x$ and $y$, s.t. $y^2=x$. True. We just need to show an example. Let's assume that $x=1$ and $y=1$. Then $1^1=1=1$. A game metaphor would be ``You pick two positive numbers $x$ and $y$. If you can show that $y^2=x$ in every game, you win''.
    \end{solution}

    \begin{exercise}
        For every positive number $y$, there exists a positive number $x$ such that $y^2 = x$.
    \end{exercise}
    \begin{solution}
        For any a positive number $y$, one can pick a positive number $x$ s.t. $y^2=x$ is true. True. \emph{(Informal Proof)}.  Since we know the value $y$, we can derive $x$ as a function of $y: x(y)=y^2$. Since the function's domain is $(0, \infty)$ for $y \in (0, \infty)$, then for any positive $y$ we can find $x(y)=y^2$. As a game metaphor ``An opponent picks a positive number $y$. You need to select a positive number $x$. If you can show every time that $y^2=x$, then you win''.
    \end{solution}

    \begin{exercise}
        There exists a positive number $y$ such that for every positive number $x$, we have $y^2 = x$.
    \end{exercise}

    \begin{solution}
        One can pick a positive number $y$ such that for any positive number $x$, the statement $y^2=x$ is true. False. Let's assume that such $y$ exists. Then we pick positive numbers $x_1$ and $x_2$ s.t. $x_1\neq x_2$. For instance, 1 and 2 respectively. If $x_1=1, x_2=4$, then $y^2=1$ and $y^2=4$. If $y^2=1$ and $y^2=4$, then $1\contreq 4$. A game metaphor in this case would be ``You pick a positive number $y$, then an opponent picks a positive number $x$. If you can show that $y^2=x$ in every game, you win''.
    \end{solution}

\section{Equality}
\begin{axiom}
    \textup{(Reflexivity)}. Given any object $x$, we have $x=x$.
    \label{ax:eq_reflexivity}
\end{axiom}
\begin{axiom}
    \textup{(Symmetry)}. Given any two objects $x$ and $y$ of the 
    same type, if $x=y$, then $y=x$.
    \label{ax:eq_symmetry}
\end{axiom}
\begin{axiom}
    \textup{(Transitivity)}. Given any three objects $x, y, z$ of the same type, if $x=y$ and $y=z$, then $x=z$.
    \label{ax:eq_transitivity}
\end{axiom}
\begin{axiom}
    \textup{(Substitution)}. Given any two objects $x$ and $y$ of the same type, if $x=y$, then $f(x)=f(y)$ for all functions or operations $f$. Similarly, for any property $P(x)$ depending on $x$, if $x=y$, then $P(x) \Leftrightarrow P(y)$.
    \label{ax:eq_substitution}
\end{axiom}
\setcounter{exercise}{0}
\begin{exercise}
    Suppose you have four real numbers $a, b, c, d$, and you know that $a=b$ and $c=d$. Use the above four axioms to deduce that $a+d=b+c$
\end{exercise}
\begin{solution}
    Let's define a function $f(x):= x + c$. By Axion~\ref{ax:eq_substitution} $f(a) = f(b) \Leftrightarrow a+c = b + c$. Let's define another function $g(x):= b + x$. Then again $g(d) = f(c) \Leftrightarrow b+d = b + c$. If $b+d=b+c$, then $b+c=b+d$ by Axiom~\ref{ax:eq_symmetry}. If $b+c=b+d$ and $b+c= a+c$, then $a+c=b+d$ by Axiom~\ref{ax:eq_transitivity}.
\end{solution}
